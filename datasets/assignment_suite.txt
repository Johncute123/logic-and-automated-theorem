# --- easy (10): list identities + propositional schemas ---
xs @ [] = xs
[] @ xs = xs
map id xs = xs
length (rev xs) = length xs
length (xs @ ys) = length xs + length ys
(A ⟶ B) ⟷ (¬A ∨ B)
(¬ A ∧ ¬ B) ⟷ ¬(A ∨ B)
(A ∧ (B ∨ C)) ⟶ ((A ∧ B) ∨ (A ∧ C))
(A ⟶ B) ⟶ (¬B ⟶ ¬A)
(if A then B else B) = B

# --- medium (10): append/map/filter + quantifiers ---

(∀x. P x ⟶ Q x) ⟶ ((∃x. P x) ⟶ (∃x. Q x))
((A ⟶ C) ∧ (B ⟶ C)) ⟶ (A ∨ B ⟶ C)
(f = g) ⟷ (∀x. f x = g x)
map f (filter p xs) = filter (λx. p x) (map f xs)
